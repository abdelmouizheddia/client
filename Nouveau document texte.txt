function serializeMember(member) {
  const latestSub = member.subscriptions?.[0];
  return {
    ...member,
    age: calculateAge(member.birthDate),
    currentPlan: latestSub?.planType || null,
    subscriptionEndDate: latestSub?.endDate || null,
  };
}

// قائمة الأعضاء مع البحث والتصفية
router.get('/', async (req, res) => {
  await refreshStatuses();
  const { search, sportId, gender, status, dateFrom, dateTo, page = 1, limit = 20 } = req.query;

  const where = { isArchived: false };

  // المدربة ترى العضوات (الإناث) فقط
  if (req.isCoach) {
    where.gender = 'female';
  } else if (gender) {
    where.gender = gender;
  }

  if (search) {
    where.OR = [
      { firstName: { contains: search } },
      { lastName: { contains: search } },
      { memberCode: { contains: search } },
      { phone: { contains: search } },
    ];
  }
  if (sportId) where.sportId = Number(sportId);
  if (status) where.status = status;
  if (dateFrom || dateTo) {
    where.registrationDate = {};
    if (dateFrom) where.registrationDate.gte = new Date(dateFrom);
    if (dateTo) where.registrationDate.lte = new Date(dateTo);
  }

  const take = Number(limit);
  const skip = (Number(page) - 1) * take;

  const [members, total] = await Promise.all([
    prisma.member.findMany({
      where,
      include: {
        sport: true,
        coach: true,
        subscriptions: { orderBy: { endDate: 'desc' }, take: 1 },
      },
      orderBy: { createdAt: 'desc' },
      skip,
      take,
    }),
    prisma.member.count({ where }),
  ]);

  res.json({
    data: members.map(serializeMember),
    total,
    page: Number(page),
    totalPages: Math.ceil(total / take) || 1,
  });
});

router.get('/:id', async (req, res) => {
  const member = await prisma.member.findUnique({
    where: { id: Number(req.params.id) },
    include: {
      sport: true,
      coach: true,
      subscriptions: { orderBy: { startDate: 'desc' } },
      attendances: { orderBy: { date: 'desc' }, take: 50 },
      achievements: { orderBy: { date: 'desc' } },
      documents: { orderBy: { createdAt: 'desc' } },
    },
  });
  if (!member) return res.status(404).json({ message: 'العضو غير موجود' });
  res.json(serializeMember(member));
});

// تعديل بيانات عضو
router.put('/:id', photoUpload.single('photo'), async (req, res) => {
  try {
    const id = Number(req.params.id);
    const {
      firstName, lastName, birthDate, gender, phone, email, address,
      sportId, coachId, coachNotes, status, rfidTag, notificationsOptIn,
    } = req.body;

    const data = { firstName, lastName, gender, phone, email, address, coachNotes };
    if (birthDate) data.birthDate = new Date(birthDate);
    if (sportId !== undefined) data.sportId = sportId ? Number(sportId) : null;
    if (coachId !== undefined) data.coachId = coachId ? Number(coachId) : null;
    if (status) data.status = status;
    if (req.file) data.photo = `/uploads/photos/${req.file.filename}`;
    if (rfidTag !== undefined) data.rfidTag = rfidTag || null;
    if (notificationsOptIn !== undefined) data.notificationsOptIn = notificationsOptIn === 'true' || notificationsOptIn === true;

    const member = await prisma.member.update({ where: { id }, data });
    res.json(member);
  } catch (err) {
    if (err.code === 'P2002' && err.meta?.target?.includes('rfidTag')) {
      return res.status(409).json({ message: 'رقم RFID هذا مسجَّل لعضو آخر. يرجى التحقق.' });
    }
    res.status(400).json({ message: 'تعذر تحديث بيانات العضو', error: err.message });
  }
});

// رفع مستندات إضافية (صور / PDF)
router.post('/:id/documents', documentUpload.array('files', 10), async (req, res) => {
  try {
    const memberId = Number(req.params.id);
    const created = [];
    for (const file of req.files || []) {
      const fileType = file.mimetype === 'application/pdf' ? 'pdf' : 'image';
      const doc = await prisma.document.create({
        data: {
          memberId,
          fileUrl: `/uploads/documents/${file.filename}`,
          fileType,
          label: req.body.label || null,
        },
      });
      created.push(doc);
    }
    res.status(201).json(created);
  } catch (err) {
    res.status(400).json({ message: 'تعذر رفع المستندات', error: err.message });
  }
});
